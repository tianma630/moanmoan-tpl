这里采用 “匿名优先” 原则，不绑定真实手机号，使用 device_id + 随机马甲 作为用户唯一标识。以下是核心表的 DDL（数据定义语言）设计。

1. 用户与马甲表
sql
-- 用户基础表（匿名ID）
CREATE TABLE `users` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '用户内部ID',
  `uuid` CHAR(36) NOT NULL DEFAULT (UUID()) COMMENT '对外匿名唯一标识',
  `device_id` VARCHAR(64) NOT NULL COMMENT '设备指纹/首次登录标识',
  `current_masthead_id` INT UNSIGNED DEFAULT NULL COMMENT '当前使用的马甲ID',
  `energy` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '吐槽能量值',
  `total_posts` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '发帖总数',
  `total_pats_received` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '收到的拍一拍总数',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `idx_uuid` (`uuid`),
  KEY `idx_device` (`device_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='匿名用户表';

-- 马甲历史/当前表
CREATE TABLE `mastheads` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
  `user_id` BIGINT UNSIGNED NOT NULL COMMENT '所属用户',
  `nickname` VARCHAR(30) NOT NULL COMMENT '如：暴躁的秋刀鱼',
  `avatar_url` VARCHAR(255) NOT NULL COMMENT '像素风头像CDN地址',
  `is_fixed` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '是否为付费固定马甲',
  `used_count` INT UNSIGNED NOT NULL DEFAULT 1 COMMENT '使用次数（声望）',
  `last_used_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_user` (`user_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户马甲表';
2. 吐槽帖子与标签
sql
-- 吐槽主帖表
CREATE TABLE `posts` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `user_id` BIGINT UNSIGNED NOT NULL COMMENT '发布者ID',
  `masthead_id` INT UNSIGNED NOT NULL COMMENT '发帖时使用的马甲ID（快照）',
  `content` TEXT NOT NULL COMMENT '吐槽文字内容',
  `images` JSON DEFAULT NULL COMMENT '图片URL数组，最多3张',
  `privacy_type` TINYINT NOT NULL DEFAULT 1 COMMENT '1-公开 2-仅树洞可见',
  `location_city` VARCHAR(20) DEFAULT NULL COMMENT '城市级别LBS（附近功能用）',
  `pat_count` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '拍一拍计数（冗余加速）',
  `echo_count` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '回声计数（冗余加速）',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '1-正常 2-被举报下架 3-违规删除',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_user` (`user_id`),
  KEY `idx_privacy_time` (`privacy_type`, `created_at` DESC),
  KEY `idx_location` (`location_city`),
  FULLTEXT INDEX `idx_content` (`content`) WITH PARSER ngram
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='吐槽主帖';

-- 情绪标签表
CREATE TABLE `tags` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(20) NOT NULL UNIQUE COMMENT '如：职场内耗',
  `icon` VARCHAR(50) DEFAULT NULL COMMENT '标签小图标代码',
  `sort_order` INT NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='情绪标签字典';

-- 帖子-标签关联表
CREATE TABLE `post_tags` (
  `post_id` BIGINT UNSIGNED NOT NULL,
  `tag_id` INT UNSIGNED NOT NULL,
  PRIMARY KEY (`post_id`, `tag_id`),
  KEY `idx_tag` (`tag_id`),
  FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE,
  FOREIGN KEY (`tag_id`) REFERENCES `tags` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='帖子标签关联';
3. 共鸣互动（拍一拍 + 回声）
sql
-- 拍一拍记录表（用于防刷和统计）
CREATE TABLE `pats` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `post_id` BIGINT UNSIGNED NOT NULL,
  `user_id` BIGINT UNSIGNED NOT NULL COMMENT '谁拍的',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_post` (`user_id`, `post_id`) COMMENT '同一用户对同一帖子只能拍一次',
  KEY `idx_post` (`post_id`),
  FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='拍一拍记录';

-- 回声（语音/表情包回复）表
CREATE TABLE `echos` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `post_id` BIGINT UNSIGNED NOT NULL,
  `user_id` BIGINT UNSIGNED NOT NULL,
  `masthead_id` INT UNSIGNED NOT NULL COMMENT '回复时的马甲快照',
  `type` TINYINT NOT NULL DEFAULT 1 COMMENT '1-语音 2-表情包',
  `resource_url` VARCHAR(255) NOT NULL COMMENT '语音文件或表情包CDN地址',
  `duration` SMALLINT UNSIGNED DEFAULT 0 COMMENT '语音时长(秒)，表情包为0',
  `text_preview` VARCHAR(100) DEFAULT NULL COMMENT '语音转文字的预览（仅用于审核展示）',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_post` (`post_id`),
  FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='回声记录';
4. 树洞信箱（核心1v1）
sql
-- 树洞会话表
CREATE TABLE `treehole_sessions` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `initiator_id` BIGINT UNSIGNED NOT NULL COMMENT '发起者',
  `receiver_id` BIGINT UNSIGNED NOT NULL COMMENT '匹配到的树洞（另一方）',
  `matched_tags` JSON DEFAULT NULL COMMENT '匹配时共同的标签，如["职场","恋爱"]',
  `expires_at` DATETIME NOT NULL COMMENT '24小时后销毁的时间点',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '1-活跃 2-已销毁 3-用户主动结束',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_initiator` (`initiator_id`),
  KEY `idx_receiver` (`receiver_id`),
  KEY `idx_expires` (`expires_at`),
  FOREIGN KEY (`initiator_id`) REFERENCES `users` (`id`) ON DELETE CASCADE,
  FOREIGN KEY (`receiver_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='树洞会话';

-- 树洞消息表（24小时后物理删除，可用定时任务清理）
CREATE TABLE `treehole_messages` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `session_id` BIGINT UNSIGNED NOT NULL,
  `sender_id` BIGINT UNSIGNED NOT NULL,
  `msg_type` TINYINT NOT NULL DEFAULT 1 COMMENT '1-文字 2-语音 3-表情包',
  `content` TEXT COMMENT '文字内容或语音转文字内容',
  `resource_url` VARCHAR(255) DEFAULT NULL COMMENT '语音/表情包资源',
  `duration` SMALLINT UNSIGNED DEFAULT 0,
  `is_read` TINYINT(1) NOT NULL DEFAULT 0,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_session` (`session_id`, `created_at`),
  FOREIGN KEY (`session_id`) REFERENCES `treehole_sessions` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='树洞消息';
5. 安全与日志表
sql
-- 敏感词触发/心理援助记录
CREATE TABLE `safety_interventions` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `user_id` BIGINT UNSIGNED NOT NULL,
  `source_type` TINYINT NOT NULL COMMENT '1-发帖 2-树洞消息',
  `source_id` BIGINT UNSIGNED NOT NULL COMMENT '对应post_id或message_id',
  `trigger_keyword` VARCHAR(50) NOT NULL COMMENT '触发的敏感词',
  `action_taken` TINYINT NOT NULL DEFAULT 1 COMMENT '1-推送关怀卡片 2-人工审核介入',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_user` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='安全干预日志';
